ELEC 475 - Lab 2: Terminal Commands
====================================
Run from the lab2 project folder with the virtual environment active:
    .\.venv\Scripts\Activate.ps1          (Windows PowerShell)
    source .venv/bin/activate             (macOS / Linux)

Datasets download automatically into ./data on first use.
All training runs below were executed on a Tesla T4 GPU (Google Colab).


--- Pre-Lab -------------------------------------------------------------------

pip install -r requirements.txt
python prelab_verify.py
python -m utils.eda --dataset cifar10
python -m utils.eda --dataset tiny_imagenet
python demo_transforms.py


--- In-Lab: LeNet-5 and Modernized LeNet (5 epochs, CIFAR-10) -----------------

python -m models.lenet
python -m models.lenet_modern

python train.py --model lenet --dataset cifar10 --epochs 5 --seed 42 --no-augment --save-model checkpoints/lenet_baseline.pth --plot-file plots/loss_lenet.png
python train.py --model lenet_modern --dataset cifar10 --epochs 5 --seed 42 --no-augment --save-model checkpoints/lenet_modern.pth --plot-file plots/loss_lenet_modern.png


--- Post-Lab Step 1: AlexNet-64 parameter verification ------------------------

python -m models.alexnet


--- Post-Lab Step 2: Normalization study (10 epochs, CIFAR-10, no augmentation)

python train.py --model alexnet --norm none --dataset cifar10 --epochs 10 --seed 42 --no-augment --save-model checkpoints/alexnet_none.pth --plot-file plots/loss_alexnet_none.png
python train.py --model alexnet --norm lrn --dataset cifar10 --epochs 10 --seed 42 --no-augment --save-model checkpoints/alexnet_lrn.pth --plot-file plots/loss_alexnet_lrn.png
python train.py --model alexnet --norm bn --dataset cifar10 --epochs 10 --seed 42 --no-augment --save-model checkpoints/alexnet_bn.pth --plot-file plots/loss_alexnet_bn.png


--- Post-Lab Step 3: Data augmentation (15 epochs, CIFAR-10) ------------------

python train.py --model alexnet --norm bn --dataset cifar10 --epochs 15 --seed 42 --save-model checkpoints/alexnet_with_aug.pth --plot-file plots/loss_with_aug.png

python train.py --model alexnet --norm bn --dataset tiny_imagenet --epochs 10 --seed 42 --save-model checkpoints/alexnet_tiny_imagenet.pth --plot-file plots/loss_tiny_imagenet.png
